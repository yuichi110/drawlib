drawlib.diagrams.architecture

High-level cloud topology and distributed system diagram engine. Combines icon-backed service cards (Node), nested network boundaries (NodeGroup), multi-way bus fan-outs (Junction), and orthogonal or curved connectors (Edge).

from drawlib.diagrams.architecture import (
    ArchitectureDiagram,
    Edge,
    GcpIcon,
    Junction,
    Node,
    NodeGroup,
    PhosphorIcon,
)
Source Code 
from drawlib.canvas import save, setup
from drawlib.diagrams.architecture import (
    ArchitectureDiagram,
    GcpIcon,
    Node,
    NodeGroup,
    PhosphorIcon,
)
from drawlib.styles import Styles

setup(width=120, height=50)

d = ArchitectureDiagram(
    node_style=Styles.Dark,
    node_text_style=Styles.DarkBold.patch(text_size=10.0),
    edge_style=Styles.Dark.patch(line_width=1.4),
    edge_text_style=Styles.Muted.patch(text_size=9.5),
    node_card_style=Styles.Neutral.patch(shape_r=2.0),
)

client = d.add(
    Node("Client", width=20, height=16, icon=PhosphorIcon.GLOBE, card_style=Styles.Neutral.patch(shape_r=2.0)),
    xy=(14, 23),
)

vpc = NodeGroup(
    title="VPC Network (NodeGroup)",
    width=80,
    height=42,
    style=Styles.MutedDashed.patch(shape_r=2.5),
    text_style=Styles.DarkBold.patch(text_size=10.5, halign="left", valign="top"),
)
d.add(vpc, xy=(34, 4))

gw = vpc.add(
    Node(
        "API Gateway",
        width=22,
        height=16,
        icon=GcpIcon.CLOUD_LOAD_BALANCING,
        card_style=Styles.PrimaryNeutral.patch(shape_r=2.0),
    ),
    xy=(18, 19),
)
app = vpc.add(
    Node("Cloud Run", width=22, height=14, icon=GcpIcon.CLOUD_RUN),
    xy=(62, 27),
)
db = vpc.add(
    Node("Cloud SQL", width=22, height=14, icon=GcpIcon.CLOUD_SQL),
    xy=(62, 10),
)

d.connect(client, gw, label="HTTPS")
gw.fork([app, db], at_x=74)

d.draw((0, 0))
save()

classArchitectureDiagram#

Container orchestrating nodes, hierarchical boundary groups, junctions, and routed edges in a unified local coordinate space.

class ArchitectureDiagram:
    def __init__(
        self,
        *,
        node_style: Style,
        node_text_style: Style,
        edge_style: Style,
        edge_text_style: Style,
        node_card_style: Style | None = None,
        title: str = "",
        title_style: Style | None = None,
        width: float | None = None,
        height: float | None = None,
        style: Style | None = None,
    ) -> None: ...
Parameter Type Default Description
node_style Style Required Default tint/fill style for vector icons inside Node cards.
node_text_style Style Required Default typography style for node labels and group titles.
edge_style Style Required Default stroke and arrowhead style for Edge connectors.
edge_text_style Style Required Default label style for connection annotations.
node_card_style Style | None None Default card background/border style (None renders borderless icon+label nodes).
title str "" Optional top-centered diagram title.
width / height float | None None Explicit canvas footprint override (auto-computed from children when None).

methodArchitectureDiagram.add(item, xy, *, show)#

Registers a Node (anchored at center xy), NodeGroup (anchored at bottom-left xy), or Junction in the diagram's coordinate frame.

def add(
    self,
    item: Node | NodeGroup | Junction,
    xy: tuple[float, float] = (0.0, 0.0),
    *,
    show: bool | None = None,
) -> Node | NodeGroup | Junction: ...

methodArchitectureDiagram.connect(start, end, ...)#

Creates and registers a routed Edge between any two connectable elements (Node, NodeGroup, or Junction).

def connect(
    self,
    start: Connectable,
    end: Connectable,
    label: str = "",
    arrow: Literal["->", "<-", "<->", "-"] = "->",
    routing: Literal["orthogonal", "direct", "curved"] = "orthogonal",
    style: Style | None = None,
    text_style: Style | None = None,
    padding: float | tuple[float, float] = 0.0,
    show: bool = True,
) -> Edge: ...

Renders all visible groups, nodes, junctions, and edges onto the active canvas.

classNode#

Represents an individual service, workload, endpoint, or datastore card with an optional icon from any of Drawlib's 34 icon suites (PhosphorIcon, GcpIcon, AwsIcon, AzureIcon, etc.).

class Node:
    def __init__(
        self,
        text: str = "",
        width: float = 20.0,
        height: float = 16.0,
        icon: IconType = None,
        icon_size: float = 8.0,
        style: Style | None = None,
        text_style: Style | None = None,
        card_style: Style | None = None,
        show: bool = True,
    ) -> None: ...

classNodeGroup#

Hierarchical boundary box (VPC, region, Kubernetes cluster, subnet) with its own bottom-left (0, 0) local coordinate frame for child .add(item, xy) calls.
