Panel

The Panel component renders titled container frames with a header band (header_variant="filled" | "underline" | "floating" | "window") and provides get_content_box(xy, width, height, padding=...) to automatically compute the inner drawable (inner_xy, inner_width, inner_height) for nesting other SmartArts, Charts, or Diagrams inside the container. Every displayed element requires an explicit Style (style, header_style, text_style always; description_style when description is non-empty; icon_style when icon is provided; dot_styles when header_variant="window").

1. Core Concepts & Basic Usage

Panel draws titled container cards anchored at the Bottom-Left (x, y) corner by default (align="bottom_left") or centered via align="center", supporting four header treatments ("filled", "underline", "floating", and "window"). Calling panel.get_content_box(xy, width, height, padding=..., content_align=...) automatically subtracts header_height and insets padding to return the safe inner (inner_xy, inner_width, inner_height) for nested components or center-anchored shapes:

def draw_step_bar(active_part: int) -> None:
    """Draw top 4-part progress bar (active_part is 1..4)."""
    pill_w = 31.0
    pill_h = 7.4
    gap = 2.0
    start_x = 4.0 + pill_w / 2.0
    cy = 75.2

    for idx, (label, sublabel) in enumerate(PART_ITEMS, start=1):
        cx = start_x + (idx - 1) * (pill_w + gap)
        if idx > 1:
            prev_cx = cx - (pill_w + gap)
            line(
                (prev_cx + pill_w / 2.0 + 0.2, cy),
                (cx - pill_w / 2.0 - 0.2, cy),
                style=Styles.PrimaryBold if idx <= active_part else Styles.MutedThin,
            )

        if idx == active_part:
            st = Styles.PrimaryFlat.patch(shape_r=1.4)
            tst = Styles.WhiteBold.patch(text_size=10.3)
            sub_tst = Styles.White.patch(text_size=9.5)
        elif idx < active_part:
            st = Styles.PrimaryNeutral.patch(shape_r=1.4, shape_line_width=1.1)
            tst = Styles.PrimaryBold.patch(text_size=10.3)
            sub_tst = Styles.Primary.patch(text_size=9.5)
        else:
            st = Styles.Neutral.patch(shape_r=1.4, shape_line_width=1.1)
            tst = Styles.MutedBold.patch(text_size=10.3)
            sub_tst = Styles.Muted.patch(text_size=9.5)

        rectangle((cx, cy), width=pill_w, height=pill_h, style=st)
        text((cx, cy + 1.35), label, style=tst)
        text((cx, cy - 1.55), sublabel, style=sub_tst)

def draw_geometry_studio(param_badge: str, footer_caption: str) -> tuple[float, float]:
    """Draw right geometry lab shell, inner coordinate sheet with subtle grid, and footer."""
    x0, w = 70.0, 64.0
    cx = x0 + w / 2.0
    top_y = 63.2
    header_h = 5.4
    body_h = 59.6

    rectangle((cx, top_y + header_h / 2.0), width=w, height=header_h, style=PANE_HEADER)
    phosphor.ruler((x0 + 3.6, top_y + header_h / 2.0), width=3.4, style=Styles.Primary)
    text(
        (x0 + 6.5, top_y + header_h / 2.0),
        "Live Geometry & Guides",
        style=Styles.DarkBold.patch(text_size=10.2, halign="left"),
    )
    text(
        (x0 + w - 2.6, top_y + header_h / 2.0),
        param_badge,
        style=Styles.PrimaryBold.patch(text_size=9.6, halign="right"),
    )

    rectangle((cx, 3.6 + body_h / 2.0), width=w, height=body_h, style=STUDIO_BODY)

    sheet_cx, sheet_cy = cx, 36.2
    sheet_w, sheet_h = 54.0, 40.0
    sx0, sy0 = sheet_cx - sheet_w / 2.0, sheet_cy - sheet_h / 2.0
    rectangle(
        (sheet_cx, sheet_cy),
        width=sheet_w,
        height=sheet_h,
        style=Styles.Neutral.patch(
            shape_fill_color=(255, 255, 255),
            shape_line_color=(180, 194, 214),
            shape_line_width=1.2,
        ),
    )
    g_st = Styles.MutedThin.patch(line_color=(232, 238, 246), line_width=0.8)
    for gx in range(1, 6):
        vx = sx0 + gx * (sheet_w / 6.0)
        line((vx, sy0), (vx, sy0 + sheet_h), style=g_st)
    for gy in range(1, 5):
        vy = sy0 + gy * (sheet_h / 5.0)
        line((sx0, vy), (sx0 + sheet_w, vy), style=g_st)

    rectangle(
        (cx, 8.4),
        width=59.2,
        height=5.2,
        style=Styles.Neutral.patch(
            shape_fill_color=(255, 255, 255),
            shape_line_color=(203, 213, 225),
            shape_line_width=1.1,
            shape_r=2.4,
        ),
        text=footer_caption,
        text_style=Styles.DarkBold.patch(text_size=9.8),
    )
    return sheet_cx, sheet_cy


def draw_sub_pills(scx: float, labels: list[str], active_idx: int) -> None:
    """Draw N-step tracker pills near the top of the studio sheet."""
    n = len(labels)
    ph = 4.2
    pgap = 0.8 if n >= 4 else 0.9
    pw = (50.4 - (n - 1) * pgap) / n
    p_start = scx - (n * pw + (n - 1) * pgap) / 2.0 + pw / 2.0
    for idx, lbl in enumerate(labels):
        px = p_start + idx * (pw + pgap)
        if idx == active_idx:
            pst = Styles.PrimaryFlat.patch(shape_r=1.2)
            ptst = Styles.WhiteBold.patch(text_size=9.5)
        elif idx < active_idx:
            pst = Styles.PrimaryNeutral.patch(shape_r=1.2, shape_line_width=1.0)
            ptst = Styles.PrimaryBold.patch(text_size=9.5)
        else:
            pst = Styles.Neutral.patch(shape_r=1.2, shape_line_width=1.0)
            ptst = Styles.Muted.patch(text_size=9.5)
        rectangle((px, 52.0), width=pw, height=ph, style=pst, text=lbl, text_style=ptst)

  • Inner & Outer Center Alignment (content_align, align): Set content_align="center" on get_content_box(...) to receive the inner (cx, cy) midpoint for center-anchored shapes/icons, or pass align="center" to .draw(...) to center the outer panel itself.

2. Inner Content Box Calculation (get_content_box)

When placing another component (Roadmap, CardList, KpiCards, ChevronProcess, or BarChart) inside a Panel, call panel.get_content_box(xy, width, height, padding=(pad_x, pad_y), align=..., content_align=...) to obtain (inner_xy, inner_width, inner_height). This automatically subtracts header_height from the top and insets all four sides by padding:

  • align: "bottom_left" | "center" = "bottom_left": Matches the align passed to panel.draw(xy, width, height, align=...).
  • content_align: "bottom_left" | "center" | None = None: Controls whether the returned inner_xy is the "bottom_left" or "center" of the inner box. When None (default), it inherits align—so get_content_box(..., align="center") returns the center coordinate ready for rectangle(inner_xy, ...) or inner_component.draw(inner_xy, ..., align="center").

4. API Summary

Constructor (Panel)

Parameter Type Default Description
text str (Required) Primary panel header title text.
description str "" Optional right-aligned subtitle or tag in the header.
style Style (Required) Outer panel body background and border style.
header_style Style (Required) Style for the header band ("filled", "window"), divider line ("underline"), or pill badge ("floating").
text_style Style (Required) Text style for the primary header title (text).
description_style Style | None None Text style for description (required when description is non-empty).
icon Callable | None None Optional icon drawing callable displayed before text.
icon_style Style | None None Style passed to icon (required when icon is provided).
dot_styles tuple[Style, Style, Style] | Style | None None Single Style or 3-tuple of Style instances for the 3 window dots (required when header_variant="window").
header_variant "filled" | "underline" | "floating" | "window" "filled" Visual layout mode of the panel header.
header_height PosFloat 8.0 Height of the header band or floating pill in canvas units.
header_align "left" | "center" "left" Horizontal alignment of the primary header title ("left" or "center").
show bool True Whether to render this panel when draw() is called.

Methods

Method Signature Description
draw (xy, width=100.0, height=60.0, *, align="bottom_left", scale=1.0) -> Self Render the panel container at xy ("bottom_left" or "center").
get_content_box (xy, width=100.0, height=60.0, *, padding=(4.0, 3.0), align="bottom_left", content_align=None, scale=1.0) -> tuple[tuple[float, float], float, float] Return (inner_xy, inner_width, inner_height) for the inner content area below the header band (content_align defaults to align when None).
