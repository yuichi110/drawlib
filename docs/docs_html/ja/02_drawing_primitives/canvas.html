キャンバスと座標系

canvas（キャンバス）は、Drawlib におけるすべての描画の基盤となる描画領域です。
「Illustration as Code（コードとしての図解）」のパラダイムのもと、キャンバスのサイズ、解像度、背景色、座標グリッド、および出力フォーマットはすべて宣言的な Python 関数呼び出しによって管理されます。 Drawlib はアクティブな canvas シングルトンを保持しており、宣言的な API 呼び出しを通じてサイズ設定、背景色と透過度、位置合わせグリッド、ローカル座標変換、マルチフォーマット出力、そして状態リセットを行います。

def draw_step_bar(active_step: int) -> None:
    """Draw top 5-step feature indicator bar (active_step is 1..5)."""
    pill_w = 24.4
    pill_h = 7.4
    gap = 2.0
    start_x = 4.0 + pill_w / 2.0
    cy = 75.2

    for idx, (label, sublabel) in enumerate(STEP_ITEMS, start=1):
        cx = start_x + (idx - 1) * (pill_w + gap)
        if idx > 1:
            prev_cx = cx - (pill_w + gap)
            line(
                (prev_cx + pill_w / 2.0 + 0.2, cy),
                (cx - pill_w / 2.0 - 0.2, cy),
                style=Styles.PrimaryBold if idx <= active_step else Styles.MutedThin,
            )

        if idx == active_step:
            st = Styles.PrimaryFlat.patch(shape_r=1.4)
            tst = Styles.WhiteBold.patch(text_size=10.2)
            sub_tst = Styles.White.patch(text_size=9.5)
        elif idx < active_step:
            st = Styles.PrimaryNeutral.patch(shape_r=1.4, shape_line_width=1.1)
            tst = Styles.PrimaryBold.patch(text_size=10.2)
            sub_tst = Styles.Primary.patch(text_size=9.5)
        else:
            st = Styles.Neutral.patch(shape_r=1.4, shape_line_width=1.1)
            tst = Styles.MutedBold.patch(text_size=10.2)
            sub_tst = Styles.Muted.patch(text_size=9.5)

        rectangle((cx, cy), width=pill_w, height=pill_h, style=st)
        text((cx, cy + 1.35), label, style=tst)
        text((cx, cy - 1.55), sublabel, style=sub_tst)


def draw_left_pane(
    active_step: int,
    code_lines: list[str],
    hl_range: tuple[int, int],
    icon_fn: object,
    summary_title: str,
    summary_line1: str,
    summary_line2: str,
) -> None:
    """Draw left Python editor pane + bottom feature takeaway strip."""
    x0, w = 4.0, 63.5
    cx = x0 + w / 2.0
    top_y = 63.2
    header_h = 5.4
    editor_h = 43.8
    takeaway_h = 15.8

    # Header bar
    rectangle((cx, top_y + header_h / 2.0), width=w, height=header_h, style=PANE_HEADER)
    code_icons.python((x0 + 3.6, top_y + header_h / 2.0), width=3.4, style=Styles.Neutral)
    text(
        (x0 + 6.5, top_y + header_h / 2.0),
        "canvas_tour.py",
        style=Styles.DarkBold.patch(text_size=10.2, halign="left"),
    )
    text(
        (x0 + w - 2.6, top_y + header_h / 2.0),
        f"Step {active_step} / 5",
        style=Styles.PrimaryBold.patch(text_size=9.8, halign="right"),
    )

    # Code editor body (y in [19.4, 63.2])
    rectangle((cx, top_y - editor_h / 2.0), width=w, height=editor_h, style=EDITOR_BODY)

    hl_start, hl_end = hl_range
    hl_top = top_y - PAD_Y - hl_start * LINE_H
    hl_bot = top_y - PAD_Y - (hl_end + 1) * LINE_H
    hl_cy = (hl_top + hl_bot) / 2.0
    hl_h = hl_top - hl_bot
    rectangle(
        (cx, hl_cy),
        width=w - 0.6,
        height=hl_h,
        style=Styles.Neutral.patch(shape_fill_color=(229, 239, 255), shape_line_width=0),
    )
    rectangle(
        (x0 + 0.55, hl_cy),
        width=0.9,
        height=hl_h,
        style=Styles.PrimaryFlat.patch(shape_line_width=0),
    )

    SourceCode.draw(
        xy=(x0, top_y),
        width=w,
        code="\n".join(code_lines),
        styles=CODE_STYLES,
        code_lang="python",
        show_linenum=True,
        align="top_left",
    )

    # Bottom takeaway strip (y in [3.6, 19.4])
    take_cy = 3.6 + takeaway_h / 2.0
    rectangle((cx, take_cy), width=w, height=takeaway_h, style=TAKEAWAY_BODY)
    circle(
        (x0 + 6.4, take_cy),
        radius=3.3,
        style=Styles.PrimaryNeutral.patch(shape_line_width=1.2),
    )
    icon_fn((x0 + 6.4, take_cy), width=3.8, style=Styles.PrimaryBold)  # type: ignore[operator]
    text(
        (x0 + 11.4, take_cy + 3.6),
        summary_title,
        style=Styles.DarkBold.patch(text_size=10.4, halign="left"),
    )
    text(
        (x0 + 11.4, take_cy - 0.2),
        summary_line1,
        style=Styles.Dark.patch(text_size=9.7, halign="left"),
    )
    text(
        (x0 + 11.4, take_cy - 3.8),
        summary_line2,
        style=Styles.PrimaryBold.patch(text_size=9.6, halign="left"),
    )

def draw_pod_widget(
    cx: float,
    cy: float,
    w: float,
    h: float,
    label: str,
    *,
    variant: str = "primary",
) -> None:
    """Draw a miniature service pod widget inside the simulated canvas."""
    rectangle(
        (cx, cy),
        width=w,
        height=h,
        style=Styles.Neutral.patch(
            shape_fill_color=(255, 255, 255, 0.92),
            shape_line_color=(148, 163, 184),
            shape_line_width=1.1,
            shape_line_style="dashed",
            shape_r=1.4,
        ),
    )
    if variant == "primary":
        hdr_st = Styles.PrimaryFlat.patch(shape_r=1.1)
        txt_st = Styles.WhiteBold.patch(text_size=9.8)
    elif variant == "secondary":
        hdr_st = Styles.SecondaryNeutral.patch(shape_r=1.1, shape_line_width=1.1)
        txt_st = Styles.DarkBold.patch(text_size=9.5)
    else:
        hdr_st = Styles.Neutral.patch(
            shape_fill_color=(241, 245, 251),
            shape_line_color=(148, 163, 184),
            shape_line_width=1.1,
            shape_r=1.1,
        )
        txt_st = Styles.DarkBold.patch(text_size=9.6)

    hdr_h = max(5.2, h * 0.40)
    hdr_cy = cy + h * 0.17
    rectangle((cx, hdr_cy), width=w * 0.86, height=hdr_h, style=hdr_st, text=label, text_style=txt_st)

    node_y = cy - h * 0.25
    dx = w * 0.24
    r = max(1.15, min(1.85, w * 0.11))
    circle((cx - dx, node_y), radius=r, style=Styles.SecondaryNeutral)
    circle((cx + dx, node_y), radius=r, style=Styles.SecondaryNeutral)
    if dx - r > 0.4:
        line((cx - dx + r, node_y), (cx + dx - r, node_y), style=Styles.DarkBold.patch(line_width=1.1))


def draw_checkerboard(cx: float, cy: float, w: float, h: float, cols: int = 16, rows: int = 9) -> None:
    """Draw alpha transparency checkerboard inside simulated canvas bounds."""
    x0 = cx - w / 2.0
    y0 = cy - h / 2.0
    tw = w / cols
    th = h / rows
    light = Styles.Neutral.patch(shape_fill_color=(255, 255, 255), shape_line_width=0)
    dark = Styles.Neutral.patch(shape_fill_color=(220, 227, 237), shape_line_width=0)
    for r_idx in range(rows):
        for c_idx in range(cols):
            st = light if (r_idx + c_idx) % 2 == 0 else dark
            rectangle(
                (x0 + (c_idx + 0.5) * tw, y0 + (r_idx + 0.5) * th),
                width=tw + 0.04,
                height=th + 0.04,
                style=st,
            )
    rectangle(
        (cx, cy),
        width=w,
        height=h,
        style=Styles.Neutral.patch(
            shape_fill_color=(0, 0, 0, 0),
            shape_line_color=(148, 163, 184),
            shape_line_width=1.3,
        ),
    )


def draw_dimension_rulers(cx: float, cy: float, w: float, h: float, w_label: str, h_label: str, accent: bool) -> None:
    """Draw width and height dimension callouts around the simulated canvas."""
    x0, x1 = cx - w / 2.0, cx + w / 2.0
    y0, y1 = cy - h / 2.0, cy + h / 2.0
    arrow_st = Styles.PrimaryBold.patch(line_width=1.5) if accent else Styles.DarkBold.patch(line_width=1.4)
    pill_st = (
        Styles.PrimaryNeutral.patch(shape_r=1.2, shape_line_width=1.1)
        if accent
        else Styles.Neutral.patch(shape_fill_color=(255, 255, 255), shape_r=1.2, shape_line_width=1.0)
    )
    txt_st = Styles.PrimaryBold.patch(text_size=9.6) if accent else Styles.DarkBold.patch(text_size=9.6)

    # Top width ruler
    ry = y1 + 3.0
    line((x0, ry), (x1, ry), arrow_head="<->", style=arrow_st)
    rectangle((cx, ry + 3.4), width=16.5, height=4.2, style=pill_st, text=w_label, text_style=txt_st)

    # Left height ruler
    rx = x0 - 2.2
    line((rx, y0), (rx, y1), arrow_head="<->", style=arrow_st)
    rectangle((rx - 4.4, cy), width=7.2, height=4.4, style=pill_st, text=h_label, text_style=txt_st)

1. インポートと基本ライフサイクル

キャンバスのライフサイクル関数およびシングルトンの canvas インスタンスは、drawlib.canvas からインポートします。

from drawlib.canvas import (
    canvas,       # Active singleton Canvas instance (provides canvas.transform())
    clear,        # Reset canvas state and configuration between drawings
    get_dimage,   # Render canvas in-memory and return a Dimage object
    save,         # Save drawing to disk (PNG, WebP, JPG, SVG, PDF)
    setup,        # Configure dimensions, grid, background color, and DPI
    show,         # Display canvas in an interactive GUI window
)

[!IMPORTANT] 連続スクリプト実行と複数画像の生成:
1 つの Python スクリプト内で複数の図解を連続生成する場合は、各画像の間に必ず clear() を呼び出してください。呼び出さない場合、前の図解で描画した要素が後続のキャンバスにそのまま残ってしまいます。

2. デカルト座標空間

Drawlib は数学的なデカルト座標系（直交座標系）を採用しています。

  • 原点 (0, 0): キャンバスの左下隅に固定されています。
  • X 軸: 右方向に向かって水平に増加します（0 〜 width）。
  • Y 軸: 上方向に向かって垂直に増加します（0 〜 height）。
  • 中心基準の配置（センターアンカー）: 図形、テキスト、アイコンに指定する座標 (x, y) は、デフォルトで要素の幾何学的な中心を表します。

3. 関数・メソッドリファレンス

3.1. setup()

キャンバスのサイズ、解像度、座標グリッド、および背景の外観を設定します。図形要素を描画する前に呼び出す必要があります。

setup(
    width: int | None = None,
    height: int | None = None,
    dpi: int | None = None,
    color: Color | tuple[int, int, int] | tuple[int, int, int, float] | str | None = None,
    alpha: float | None = None,
    background_color: Color | tuple[int, int, int] | tuple[int, int, int, float] | str | None = None,
    background_alpha: float | None = None,
    grid: bool | None = None,
    grid_only: bool | None = None,
    grid_style: Style | None = None,
    grid_centerstyle: Style | None = None,
    grid_xpitch: int | None = None,
    grid_ypitch: int | None = None,
)

パラメータリファレンス:

パラメータ 型 デフォルト 説明
width int 100 仮想座標単位でのキャンバスの幅（> 0）。
height int 100 仮想座標単位でのキャンバスの高さ（> 0）。
dpi int 100 1 インチあたりのドット数（解像度およびラスター画像の鮮明さ）。
background_color / color Color | tuple | str (255, 255, 255) キャンバスの背景色。Color インスタンス、RGB/RGBA タプル、16 進カラーコード文字列（"#f8fafc"）、または CSS 色名を指定できます（color は background_color の短縮エイリアスです）。
background_alpha / alpha float 1.0 [0.0, 1.0] の範囲で指定する背景の不透明度（0.0 = 完全透明、1.0 = 不透明。alpha は background_alpha の短縮エイリアスです）。
grid bool False レイアウト位置合わせ用の座標グリッド線を重ねて描画します。
grid_only bool False グリッドなしの通常画像を別途保存せず、座標グリッドを重ねた画像のみを出力します。
grid_style Style 灰色の破線 通常のグリッド線のカスタム線スタイル（指定すると自動的に grid=True が有効になります）。
grid_centerstyle Style 太い灰色の破線 中心軸（x = width / 2, y = height / 2）のカスタム線スタイル。
grid_xpitch int width / 10 垂直グリッド線の間隔。
grid_ypitch int height / 10 水平グリッド線の間隔。

3.2. save()

アクティブなキャンバスをディスク上の画像ファイルとして書き出します。

save(
    file: str | None = None,
    format: Literal["png", "webp", "jpg", "pdf", "svg"] | None = None,
)
  • file: 出力先のファイルパス。省略した場合は、実行中スクリプトと同じディレクトリ内の <script_name>.png がデフォルトになります。
  • format: 出力画像フォーマット（"png", "webp", "jpg"/"jpeg", "svg", "pdf"）。省略した場合はファイル名の拡張子から推論され、拡張子もない場合は "png" になります。
  • グリッド有効時の出力動作 (setup(..., grid=True)):
    • grid=True かつ grid_only=False（デフォルト）の場合、save("out.png") を呼び出すと、グリッドなしのクリーンな図解（out.png）とグリッド重ね合わせ画像（out_grid.png）の 2 つのファイルが出力されます。
    • grid_only=True の場合、save("out.png") を呼び出すと、座標グリッドが直接重ねられた 1 つのファイル（out.png）のみが出力されます。

[!NOTE] Markdown コードブロック内での save() の扱い: Markdown 埋め込みコードブロック（```drawlib）では、ドキュメントビルダが自動的にキャンバスをキャプチャするため、save() の呼び出しは任意です。埋め込みブロック内で save() が呼び出された場合でも、エンジンは重複書き込みや競合を防ぐため安全に no-op（何もしない処理）として扱います。ただし、スタンドアロンの .py スクリプトへそのままコピー＆ペーストして動作させられるよう、完全なサンプルコードでは引数なしの save() を明記することを推奨します。

3.3. show()

ローカルの対話型デスクトップウィンドウを開き、図解を表示します。ヘッドレス環境（CI/CD、Docker、リモートサーバーなど）では、代わりに save() または CLI の drawlib show -o <path> を使用してください。

3.4. clear()

キャンバスのジオメトリをリセットし、登録済みの描画要素（アーティスト）を消去して、デフォルトのスタイルと設定パラメータを復元します。

3.5. get_dimage()

キャンバスをメモリ上でレンダリングして Dimage オブジェクトを返します。ディスクに一時ファイルを書き出すことなく、プログラム上で画像の変換・トリミング・合成処理をつなげることができます。

3.6. ローカル座標のスケーリングと平行移動 (canvas.transform())

canvas.transform() コンテキストマネージャは、with ブロック内で描画されるすべての図形・線・アイコン・テキストに対して相似変換（基準点 origin を中心とした比例スケーリングと (dx, dy) の平行移動オフセット）を適用します。変換ブロックはネストさせることも可能です。

with canvas.transform(
    origin: tuple[float, float] = (0.0, 0.0),
    scale: float = 1.0,
    translate: tuple[float, float] = (0.0, 0.0),
):
    ...
  • origin: スケーリングの中心となる基準座標 (ox, oy)（デフォルトは (0.0, 0.0)）。
  • scale: 正の比例スケール係数（> 0、デフォルトは 1.0）。図形サイズ、線幅、矢印ヘッド、フォントサイズを一括でスケーリングします。
  • translate: キャンバス単位で適用される追加の平行移動オフセット (dx, dy)（デフォルトは (0.0, 0.0)）。
